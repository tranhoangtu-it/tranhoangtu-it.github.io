// Triggered by ESP32-S3 via HTTPS/MQTT when Hall Lid Sensor stays CLOSED > 20 mins
export async function handleMissedDoseAlert(event, twilioClient) {
  const { patientName, slotLabel, medication, scheduledTime, caregivers } = event;
  const msg = `[MedBox Urgent] ${patientName} missed the ${scheduledTime} dose (${slotLabel}: ${medication}) after 2 loud voice + vibration alerts.`;

  for (const cg of caregivers) {
    // 1. Instant SMS / WhatsApp text alert
    await twilioClient.messages.create({
      to: cg.phone,
      from: process.env.TWILIO_NUMBER,
      body: `${msg} Reply SNOOZE to delay 15m or call ${patientName} now.`
    });

    // 2. Automated TwiML Spoken Phone Call to Family Caregiver
    await twilioClient.calls.create({
      to: cg.phone,
      from: process.env.TWILIO_NUMBER,
      twiml: `<Response>
        <Say voice="Polly.Aditi" language="en-IN">
          Urgent Medication Alert. ${patientName} has not opened ${slotLabel} for ${medication}, scheduled at ${scheduledTime}. Press 1 to speak directly through the medicine box speakerphone.
        </Say>
      </Response>`
    });
  }
  return { escalated: true, notifiedCount: caregivers.length };
}
